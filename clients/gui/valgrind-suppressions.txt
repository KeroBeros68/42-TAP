# **************************************************************************** #
#                       Valgrind suppression file (42-TAP GUI)                 #
#                                                                              #
# Hides leaks that come from system / third-party libraries (Qt, X11, glibc,   #
# libstdc++, fontconfig, ALSA, ...) so that only leaks caused by our own code  #
# are reported. Regenerate entries with:                                       #
#   valgrind --leak-check=full --show-leak-kinds=all --gen-suppressions=all     #
#   ./42TAP                                                                   #
# and copy the interesting blocks here (name: first line, frames below).       #
# **************************************************************************** #

{
   Qt library allocations
   Memcheck:Leak
   match-leak-kinds: definite,indirect,possible,reachable
   fun:*alloc
   ...
   obj:*libQt5*
}
